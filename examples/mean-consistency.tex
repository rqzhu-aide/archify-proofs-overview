\title{Sample-mean consistency: a synthetic teaching example}

\maketitle
This is an illustrative example, not a published paper.
The following two assumptions are in force throughout.
\begin{assumption}\label{ass:sampling}
$X_1,X_2,\ldots$ are independent and identically distributed.
\end{assumption}
\begin{assumption}\label{ass:moments}
$\mathbb{E}X_i=\mu$ and $\operatorname{Var}(X_i)=\sigma^2<\infty$.
\end{assumption}
\begin{definition}\label{def:mean}
The sample mean is $\bar X_n=n^{-1}\sum_{i=1}^{n}X_i$.
\end{definition}
\begin{lemma}\label{lem:centering}
$\mathbb{E}\bar X_n=\mu$.
\end{lemma}
\begin{proof}
By Definition~\ref{def:mean}, linearity of expectation and
Assumption~\ref{ass:moments}, the expectation is $n^{-1}n\mu=\mu$.
\end{proof}
\begin{lemma}\label{lem:variance}
$\operatorname{Var}(\bar X_n)=\sigma^2/n$.
\end{lemma}
\begin{proof}
By Assumption~\ref{ass:sampling}, distinct summands have zero covariance.
Definition~\ref{def:mean} and Assumption~\ref{ass:moments} therefore give
$\operatorname{Var}(\bar X_n)=n^{-2}\sum_{i=1}^n\sigma^2=\sigma^2/n$.
\end{proof}
\begin{theorem}\label{thm:bound}
For every $\varepsilon>0$,
$\Pr(|\bar X_n-\mu|>\varepsilon)\leq\sigma^2/(n\varepsilon^2)$.
\end{theorem}
\begin{proof}
Chebyshev's inequality states that, for finite-variance $Y$ and
$\varepsilon>0$, $\Pr(|Y-\mathbb{E}Y|>\varepsilon)\leq\operatorname{Var}(Y)/\varepsilon^2$.
Apply it with $Y=\bar X_n$, and substitute Lemmas~\ref{lem:centering}
and~\ref{lem:variance} to obtain the bound.
\end{proof}
\begin{corollary}\label{cor:consistency}
$\bar X_n\xrightarrow{p}\mu$ as $n\to\infty$.
\end{corollary}
\begin{proof}
For each fixed $\varepsilon>0$, the bound in Theorem~\ref{thm:bound}
tends to zero as $n\to\infty$, which is convergence in probability.
\end{proof}
